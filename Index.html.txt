    pollInterval = setInterval(function() {

  function handleAdminKill(graceSeconds) {

    if ($('admin-kill-notice').classList.contains('show')) return;

   

    logSuspicious('admin_kill', 'Exam ended by administrator. Grace: ' + graceSeconds + 's');

   

    if (graceSeconds > 0) {

      killGraceRemaining = graceSeconds;

      $('admin-kill-timer').textContent = killGraceRemaining;

      $('admin-kill-sec').textContent = killGraceRemaining;

      $('admin-kill-notice').classList.add('show');focusFirstInDialog('admin-kill-notice');

     

      killGraceTimer = setInterval(function() {

        killGraceRemaining--;

        $('admin-kill-timer').textContent = killGraceRemaining;

        $('admin-kill-sec').textContent = killGraceRemaining;

       

        if (killGraceRemaining <= 0) {

          clearInterval(killGraceTimer);

          $('admin-kill-notice').classList.remove('show');

          submitExam(true,'administrator');

        }

      }, 1000);

    } else {

      submitExam(true,'administrator');

    }

  }

  function showSubmitConfirm(){

  function formatReviewAnswer(item,value,isCorrectAnswer){
    if(item.type==='fillblank'){
      if(isCorrectAnswer&&Array.isArray(value)){
        return value.map(group=>Array.isArray(group)?group.join(' / '):String(group||'')).filter(Boolean).join(' ; ')||'No answer';
      }
      if(Array.isArray(value))return value.map(v=>String(v||'').trim()).filter(Boolean).join(' / ')||'No answer';
      return value!==null&&value!==undefined&&String(value).trim()?String(value):'No answer';
    }
    if(item.type==='rcoding'){
      if(isCorrectAnswer)return value!==null&&value!==undefined&&String(value).trim()?String(value):'Assessed using the configured automated tests.';
      return value!==null&&value!==undefined&&String(value).trim()?String(value):'No answer';
    }
    const indexes=(Array.isArray(value)?value:[value]).map(Number).filter(index=>Number.isInteger(index)&&index>=0&&index<(item.options||[]).length);
    return indexes.length?indexes.map(index=>LETTERS[index]+'. '+item.options[index]).join(' / '):'No answer';
  }

  }
