            document.documentElement.dataset.theme = mq.matches ? 'dark' : 'light';
          } catch (e) {}
        };
        if (mq.addEventListener) mq.addEventListener('change', sync);
      }

      var copyBtn = document.getElementById('copy-citation-btn');
      if (copyBtn) {
        var citationPlain = 'Bagchi, D. (2026). Doctor Proctor: Software for data collection [Computer software]. Zenodo. https://doi.org/10.5281/zenodo.24419500';
        var label = copyBtn.querySelector('span');
        var resetTimer;

        var showCopied = function () {
          copyBtn.classList.add('copied');
          if (label) label.textContent = 'Copied';
          clearTimeout(resetTimer);
          resetTimer = setTimeout(function () {
            copyBtn.classList.remove('copied');
            if (label) label.textContent = 'Copy citation';
          }, 2000);
        };

        copyBtn.addEventListener('click', function () {
          if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(citationPlain).then(showCopied).catch(function () {
              fallbackCopy(citationPlain, showCopied);
            });
          } else {
            fallbackCopy(citationPlain, showCopied);
          }
        });

        function fallbackCopy(text, onSuccess) {
          var ta = document.createElement('textarea');
          ta.value = text;
          ta.setAttribute('readonly', '');
          ta.style.position = 'fixed';
          ta.style.opacity = '0';
          document.body.appendChild(ta);
          ta.select();
          try {
            if (document.execCommand('copy')) onSuccess();
          } catch (e) {}
          document.body.removeChild(ta);
        }
      }
    })();
  </script>
