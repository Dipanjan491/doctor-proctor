  window.webR = null;

  window.initWebR = async function() {
    if (window.webRReady && window.webR) return window.webR;
    if (window.webRInitPromise) return window.webRInitPromise;
    window.webRInitPromise = (async function(){
      try {
        const { WebR } = await import('https://webr.r-wasm.org/latest/webr.mjs');
        const instance = new WebR();
        await instance.init();
        window.webR = instance;
        window.webRReady = true;
        return instance;
      } catch (error) {
        window.webR = null;
        window.webRReady = false;
        window.webRInitPromise = null;
        throw error;
      }
    })();

  function initCopyPasteBlock() {

    if (ALLOW_COPY_PASTE) return;

    const blockEvents = ['copy', 'cut', 'paste', 'dragstart', 'drop', 'contextmenu'];

   

    blockEvents.forEach(evt => {

      document.addEventListener(evt, function(e) {

        if (!examActive) return;

       

        // Allowed inside R Coding blocks specifically

        if (e.target && e.target.closest && (e.target.closest('.CodeMirror') || e.target.closest('.code-editor'))) return;

        e.preventDefault();

        e.stopPropagation();

       

        const target = e.target.closest('.opt, .blank-input, .q-text, .btn-nav, .btn-primary') || document.body;

        target.classList.remove('copy-blocked');

        void target.offsetWidth;

        target.classList.add('copy-blocked');

        setTimeout(() => target.classList.remove('copy-blocked'), 2000);

       

        logSuspicious('copy_paste_attempt', evt);

        return false;

      }, true);

    });

   

    document.addEventListener('keydown', function(e) {

      if (!examActive) return;
      const inCodeEditor=!!(e.target && e.target.closest && (e.target.closest('.CodeMirror') || e.target.closest('.code-editor')));
      const key = e.key.toLowerCase();
      const ctrl = e.ctrlKey || e.metaKey;
      if(inCodeEditor && ctrl && ['c','x','v','a'].includes(key))return;

     

      if (ctrl && ['c','x','v','a','p','s'].includes(key)) {

        e.preventDefault();

        e.stopPropagation();

        logSuspicious('keyboard_shortcut_blocked', 'Ctrl/Cmd+' + key.toUpperCase());

        return false;

      }

     

      if (key === 'printscreen' || key === 'prtsc' || key === 'snapshot') {

        e.preventDefault();

        document.body.classList.add('screenshot-blur');

        setTimeout(() => document.body.classList.remove('screenshot-blur'), 1500);

        logSuspicious('screenshot_attempt', 'PrintScreen key');

        return false;

      }

    }, true);

   

    const noSelectCSS = `

      .q-text, .opt, .opt-text, .part-label, .q-counter {

        -webkit-user-select: none !important;

        -moz-user-select: none !important;

        -ms-user-select: none !important;

        user-select: none !important;

      }

      .blank-input, .code-editor, .r-output {

        -webkit-user-select: text !important;

        -moz-user-select: text !important;

        -ms-user-select: text !important;

        user-select: text !important;

      }

    `;

    const style = document.createElement('style');

    style.textContent = noSelectCSS;

    document.head.appendChild(style);

   

    document.addEventListener('keydown', function(e) {

      if (!examActive) return;

      if (e.key === 'F12') {

        e.preventDefault();

        logSuspicious('devtools_attempt', 'F12');

        return false;

      }

      if ((e.ctrlKey && e.shiftKey && ['i','j','c','k'].includes(e.key.toLowerCase())) ||

          (e.metaKey && e.altKey && ['i','j'].includes(e.key.toLowerCase()))) {

        e.preventDefault();

        e.stopPropagation();

        logSuspicious('devtools_attempt', 'Ctrl+Shift+' + e.key.toUpperCase());

        return false;

      }

    }, true);

  }

  function normalizeText(s){return String(s||'').trim().toLowerCase().replace(/\s+/g,' ');}

  function isMcqCorrect(userAnswer,q){

    const expected=(Array.isArray(q.correct)?q.correct:[q.correct]).map(Number).sort();

    const chosen=(Array.isArray(userAnswer)?userAnswer:[userAnswer]).filter(value=>value!==null&&value!==undefined&&value!=='').map(Number).sort();

    return expected.length>0&&expected.length===chosen.length&&expected.every((value,index)=>value===chosen[index]);

  }

  function getBlankCount(q){

    if(q.type!=='fillblank')return 1;

    if(q.blanks>=1)return q.blanks;

    if(Array.isArray(q.correct)&&Array.isArray(q.correct[0]))return q.correct.length;

    return 1;

  }

  function isFillBlankCorrect(userAnswer,q){

    const n=getBlankCount(q);

    if(n>1){

      if(!Array.isArray(userAnswer))return false;

      const groups=q.correct;

      for(let i=0;i<groups.length;i++){

        const ua=userAnswer[i];

        if(ua===null||ua===undefined||!String(ua).trim())return false;

        const uaN=normalizeText(ua);

        if(!groups[i].some(a=>normalizeText(a)===uaN))return false;

      }

      return true;

    }

    if(userAnswer===null||userAnswer===undefined||!String(userAnswer).trim())return false;

    const accepted=Array.isArray(q.correct)?q.correct:[q.correct];

    const ua=normalizeText(userAnswer);

    return accepted.some(a=>normalizeText(a)===ua);

  }

  function formatReviewAnswer(item,value,isCorrectAnswer){

      }
      if(Array.isArray(value))return value.map(v=>String(v||'').trim()).filter(Boolean).join(' / ')||'No answer';
      return value!==null&&value!==undefined&&String(value).trim()?String(value):'No answer';
    }
    if(item.type==='rcoding'){
      if(isCorrectAnswer)return value!==null&&value!==undefined&&String(value).trim()?String(value):'Assessed using the configured automated tests.';
      return value!==null&&value!==undefined&&String(value).trim()?String(value):'No answer';
    }
    const indexes=(Array.isArray(value)?value:[value]).map(Number).filter(index=>Number.isInteger(index)&&index>=0&&index<(item.options||[]).length);
    return indexes.length?indexes.map(index=>LETTERS[index]+'. '+item.options[index]).join(' / '):'No answer';
  }
