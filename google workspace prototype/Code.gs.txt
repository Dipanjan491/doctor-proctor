function doGet(e) {


  e = e || {};





  var params = e.parameter || {};







  var page = String(params.page || (params.embedded === "true" ? "exam" : "home")).toLowerCase();







  var webAppUrl = ScriptApp.getService().getUrl() || "";







  if (page === "admin") {







    var template = HtmlService.createTemplateFromFile("Admin");







    template.webAppUrl = webAppUrl;







    return template.evaluate()







      .setTitle("Doctor Proctor | Admin")







      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)







      .addMetaTag("viewport", "width=device-width, initial-scale=1");







  }







  if (page === "exam") {







  // Student exam page -- load active question set from sheet







  var activeSet = getActiveQuestionSet();







  var template  = HtmlService.createTemplateFromFile("Index");







  template.webAppUrl     = webAppUrl;







  template.examTitle     = activeSet.title     || "MCQ Exam";







  template.examSubtitle  = activeSet.subtitle  || "";







  template.timeLimit     = activeSet.timeLimit || 2700;







  template.questionsJson = JSON.stringify(activeSet.questions || []);







  template.sharedDatasetJson = JSON.stringify(activeSet.sharedDataset || null);







  template.hasQuestions  = (activeSet.questions && activeSet.questions.length > 0);







  template.allowCopyPaste = !!activeSet.allowCopyPaste;







  // Privacy-first: the review is available only when explicitly enabled.



  // Older saved exams without this setting therefore keep answers hidden.



  template.showAnswerReview = isAnswerReviewEnabled(activeSet);

  template.randomizeQuestions = activeSet.randomizeQuestions === true;















  return template.evaluate()







    .setTitle((activeSet.title || "MCQ Exam") + " | Doctor Proctor")







    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)







    .addMetaTag("viewport", "width=device-width, initial-scale=1");







  }







  // Default: branded Doctor Proctor landing page







  var landingSet = getActiveQuestionSet();







  var homeTemplate = HtmlService.createTemplateFromFile("Home");







  homeTemplate.webAppUrl = webAppUrl;







  homeTemplate.hasActiveExam = !!(landingSet.questions && landingSet.questions.length);







  homeTemplate.activeExamTitle = landingSet.title || "Current Exam";







  return homeTemplate.evaluate()







    .setTitle("Doctor Proctor | Secure Online Assessments")







    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)







    .addMetaTag("viewport", "width=device-width, initial-scale=1");







}







function isAnswerReviewEnabled(examSet) {







  return !!(examSet && examSet.showAnswerReview === true);







}



// Read the current server-side preference at submission time. The dedicated

// script property is written whenever an exam is launched or updated. Only

// the exact string "true" reveals answers; every other value fails closed.

function getAnswerReviewSetting() {



  try {



    var stored = PropertiesService.getScriptProperties().getProperty("SHOW_ANSWER_REVIEW");



    if (stored !== null) return stored === "true";



    // Compatibility fallback for an active exam created before this fix.

    return isAnswerReviewEnabled(getActiveQuestionSet());



  } catch (e) {



    return false;



  }



}

function ensureResponseHeaders_(sheet) {







  var headers = ["Timestamp","Name","Exam Title","Score","Correct","Total","Time Taken (s)","Answers (JSON)","Grading (JSON)","Final Score"];







  if (sheet.getMaxColumns() < headers.length) {







    sheet.insertColumnsAfter(sheet.getMaxColumns(), headers.length - sheet.getMaxColumns());

  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);







}

function getSuspiciousActivitySummary() {







  try {







    var ss = getOrCreateSheet();







    var sheet = ss.getSheetByName("SuspiciousActivity");







    if (!sheet) return [];















    var rows = sheet.getDataRange().getValues();







    if (rows.length <= 1) return [];







    var students = {};















    for (var i = 1; i < rows.length; i++) {







      var r = rows[i];







      var studentName = r[1] || "Anonymous";







      var examTitle = r[2] || "";







      var eventType = r[3] || "";







      var detail = r[4] || "";







      var questionIndex = r[5] !== "" ? parseInt(r[5]) : undefined;







      var timeLeft = r[6] !== "" ? parseInt(r[6]) : undefined;







      var timestamp = r[0] ? new Date(r[0]).toISOString() : new Date().toISOString();















      var key = studentName + "|" + examTitle;















      if (!students[key]) {







        students[key] = {







          studentName: studentName,







          examTitle: examTitle,







          events: [],







          copyCount: 0,







          tabCount: 0,







          devtoolsCount: 0,







          screenshotCount: 0,







          otherCount: 0,







          totalEvents: 0







        };







      }















      students[key].events.push({







        timestamp: timestamp,







        type: eventType,







        detail: detail,







        questionIndex: questionIndex,







        timeLeft: timeLeft







      });







      students[key].totalEvents++;















      if (eventType.indexOf('copy') >= 0 || eventType.indexOf('paste') >= 0) {







        students[key].copyCount++;







      } else if (eventType.indexOf('tab') >= 0 || eventType.indexOf('fullscreen') >= 0 || eventType.indexOf('pagehide') >= 0) {







        students[key].tabCount++;







      } else if (eventType.indexOf('dev') >= 0) {







        students[key].devtoolsCount++;







      } else if (eventType.indexOf('screenshot') >= 0) {







        students[key].screenshotCount++;







      } else {







        students[key].otherCount++;







      }







    }















    var result = [];







    for (var key in students) {







      var s = students[key];







      var riskScore = 0;







      riskScore += s.copyCount * 2;







      riskScore += s.tabCount * 3;







      riskScore += s.devtoolsCount * 4;







      riskScore += s.screenshotCount * 3;







      riskScore += s.otherCount * 1;















      var riskLevel = 'low';







      if (riskScore >= 15) riskLevel = 'high';







      else if (riskScore >= 5) riskLevel = 'medium';















      s.riskScore = riskScore;







      s.riskLevel = riskLevel;















      s.events.sort(function(a, b) {







        return new Date(a.timestamp) - new Date(b.timestamp);







      });















      result.push(s);







    }















    result.sort(function(a, b) { return b.riskScore - a.riskScore; });















    return result;







  } catch(e) {







    return [];







  }







}

function checkExamActive() {







  try {







    var ss = getOrCreateSheet();







    var controlSheet = ss.getSheetByName("ExamControl");







    if (!controlSheet) return { active: true, killed: false };















    var data = controlSheet.getRange(1, 1, 1, 4).getValues()[0];







    var killed = data[0] === true || data[0] === "true";







    var graceSeconds = parseInt(data[1]) || 0;







    var killedAt = data[2] ? new Date(data[2]).getTime() : 0;







    var reason = data[3] || "";















    if (killed && graceSeconds > 0) {







      var elapsed = (new Date().getTime() - killedAt) / 1000;







      if (elapsed >= graceSeconds) {







        controlSheet.getRange(1, 1).setValue("false");







        return { active: false, killed: false };







      }







      var remaining = Math.max(0, Math.ceil(graceSeconds - elapsed));







      return { active: true, killed: true, graceSeconds: remaining, reason: reason };







    }















    return { active: !killed, killed: killed, graceSeconds: 0, reason: reason };







  } catch(e) {







    return { active: true, killed: false };







  }







}

// Fetch & Clear Participant Responses







// ============================================================







function getParticipantResponses() {







  try {







    var ss = getOrCreateSheet();







    var sheet = ss.getSheetByName("Responses");







    if (!sheet) return [];







    ensureResponseHeaders_(sheet);















    var rows = sheet.getDataRange().getValues();







    if (rows.length <= 1) return [];







    // Older submissions did not store R test definitions inside each answer.







    // Recover them from the active or saved question set before grading.







    var availableSets = [];







    var activeSet = getActiveQuestionSet();







    if (activeSet && Array.isArray(activeSet.questions)) availableSets.push(activeSet);







    var savedSets = getSavedSets();







    for (var si = 0; si < savedSets.length; si++) {







      if (savedSets[si] && savedSets[si].data && Array.isArray(savedSets[si].data.questions)) {







        availableSets.push(savedSets[si].data);







      }







    }















    var results = [];







    for (var i = 1; i < rows.length; i++) {







      var r = rows[i];







      var answers = [];







      if (r[7]) {







        try { answers = JSON.parse(r[7]); } catch(e) { answers = []; }







      }







      var responseTitle = String(r[2] || "").trim();







      var matchingSet = null;







      for (var setIndex = availableSets.length - 1; setIndex >= 0; setIndex--) {







        if (String(availableSets[setIndex].title || "").trim() === responseTitle) {







          matchingSet = availableSets[setIndex];







          break;







        }







      }







      var definitionQuestions = matchingSet && Array.isArray(matchingSet.questions) ? matchingSet.questions : [];







      for (var ai = 0; ai < answers.length; ai++) {







        var oldAnswer = answers[ai] || {};







        if (oldAnswer.type !== "rcoding" && oldAnswer.type !== "r coding") continue;







        if (Array.isArray(oldAnswer.testDefinitions) && oldAnswer.testDefinitions.length) continue;







        var answerQuestion = String(oldAnswer.question || oldAnswer.q || "").trim();







        var questionDefinition = definitionQuestions[ai] || null;







        if (!questionDefinition || String(questionDefinition.q || "").trim() !== answerQuestion) {







          questionDefinition = null;







          for (var dqi = 0; dqi < definitionQuestions.length; dqi++) {







            if (String(definitionQuestions[dqi].q || "").trim() === answerQuestion) {







              questionDefinition = definitionQuestions[dqi];







              break;







            }







          }







        }







        if (!questionDefinition) {







          // Last fallback: find the same question text in any saved/active set.







          for (var asi = availableSets.length - 1; asi >= 0 && !questionDefinition; asi--) {







            var candidateQuestions = availableSets[asi].questions || [];







            for (var cqi = 0; cqi < candidateQuestions.length; cqi++) {







              if (String(candidateQuestions[cqi].q || "").trim() === answerQuestion) {







                questionDefinition = candidateQuestions[cqi];







                break;







              }







            }







          }







        }







        if (questionDefinition && Array.isArray(questionDefinition.tests)) {







          oldAnswer.testDefinitions = questionDefinition.tests;







        }







      }







      results.push({







        rowNumber: i + 1,







        timestamp: r[0] ? new Date(r[0]).toLocaleString() : "",







        timestampIso: r[0] && !isNaN(new Date(r[0]).getTime()) ? new Date(r[0]).toISOString() : "",







        name: r[1] || "Anonymous",







        examTitle: r[2] || "",







        score: r[3] || "0%",







        correct: r[4] || 0,







        total: r[5] || 0,







        timeTaken: r[6] || 0,







        answers: answers,







        grading: r[8] ? safeJsonParse_(r[8], {}) : {},







        finalScore: r[9] === "" || r[9] === null || r[9] === undefined ? null : r[9]







      });







    }







    return results.reverse();







  } catch(e) {







    return [];







  }







}







function safeJsonParse_(value, fallback) {







  try { return JSON.parse(value); } catch(e) { return fallback; }







}







function normalizeAnswerText_(value) {







  return String(value === null || value === undefined ? "" : value).trim().toLowerCase().replace(/\s+/g, " ");







}







function answerTextFromKey_(q) {







  if (!q) return "";







  if (q.type === "mcq") {







    var letters = ["A", "B", "C", "D"];







    var correctIndexes = Array.isArray(q.correct) ? q.correct : [q.correct];







    return correctIndexes.filter(function(index) {







      return index !== null && index !== undefined && Number(index) >= 0 && Number(index) < 4;







    }).map(function(index) {







      return "Option " + letters[Number(index)];







    }).join(" / ");







  }







  if (q.type === "fillblank") {







    if (Array.isArray(q.correct) && Array.isArray(q.correct[0])) {







      return q.correct.map(function(group, i) {







        return "Blank " + (i + 1) + ": " + (Array.isArray(group) ? group.join(" / ") : group);







      }).join(" ; ");







    }







    return Array.isArray(q.correct) ? q.correct.join(" / ") : String(q.correct || "");







  }







  return q.correct || "";







}







function isObjectiveAnswerCorrect_(userAnswer, q) {







  if (!q || q.type === "rcoding" || q.type === "r coding") return false;







  if (q.type === "fillblank") {







    if (Array.isArray(q.correct) && Array.isArray(q.correct[0])) {







      if (!Array.isArray(userAnswer)) return false;







      for (var i = 0; i < q.correct.length; i++) {







        var ua = normalizeAnswerText_(userAnswer[i]);







        if (!ua) return false;







        var accepted = q.correct[i] || [];







        var matched = accepted.some(function(a) { return normalizeAnswerText_(a) === ua; });







        if (!matched) return false;







      }







      return true;







    }







    var single = normalizeAnswerText_(userAnswer);







    if (!single) return false;







    var acceptedSingle = Array.isArray(q.correct) ? q.correct : [q.correct];







    return acceptedSingle.some(function(a) { return normalizeAnswerText_(a) === single; });







  }







  var expected = (Array.isArray(q.correct) ? q.correct : [q.correct]).map(Number).sort();







  var chosen = (Array.isArray(userAnswer) ? userAnswer : [userAnswer])







    .filter(function(value) { return value !== null && value !== undefined && value !== ""; })







    .map(Number).sort();







  if (chosen.length !== expected.length) return false;







  for (var ci = 0; ci < expected.length; ci++) {







    if (chosen[ci] !== expected[ci]) return false;







  }







  return expected.length > 0;







}







function recalculateScoresFromActiveKey() {







  try {







    var ss = getOrCreateSheet();







    var responseSheet = ss.getSheetByName("Responses");







    if (!responseSheet || responseSheet.getLastRow() <= 1) return { success: true, updated: 0 };







    ensureResponseHeaders_(responseSheet);







    var activeSet = getActiveQuestionSet();







    var keyQuestions = Array.isArray(activeSet.questions) ? activeSet.questions : [];







    if (!keyQuestions.length) return { success: false, error: "No active question set found. Launch or save the corrected exam first." };







    var activeTitle = String(activeSet.title || "").trim();







    if (!activeTitle) return { success: false, error: "The active exam has no title. Add a title and launch it again before recalculating scores." };







    var rows = responseSheet.getDataRange().getValues();







    var updated = 0;







    var skipped = 0;







    for (var r = 1; r < rows.length; r++) {







      var row = rows[r];







      if (activeTitle && String(row[2] || "").trim() !== activeTitle) {







        skipped++;







        continue;







      }







      var answers = row[7] ? safeJsonParse_(row[7], []) : [];







      if (!Array.isArray(answers)) answers = [];







      var objectiveCorrect = 0;







      var objectiveTotal = 0;







      var automatedREarned = 0;







      var automatedRMax = 0;







      for (var i = 0; i < keyQuestions.length; i++) {







        var key = keyQuestions[i] || {};







        var answer = answers[i] || {};







        if (key.type === "rcoding" || key.type === "r coding") {







          if (!answers[i]) answers[i] = {};







          if (answers[i]) {







            answers[i].correct = key.correct;







            answers[i].correctText = answerTextFromKey_(key);







            answers[i].testDefinitions = Array.isArray(key.tests) ? key.tests : [];







          }







          var definitions = Array.isArray(key.tests) ? key.tests : [];







          var results = answer.automated && Array.isArray(answer.automated.tests) ? answer.automated.tests : [];







          for (var ti = 0; ti < definitions.length; ti++) {







            var testMarks = Number(definitions[ti].marks) || 0;







            automatedRMax += testMarks;







            if (results[ti] && results[ti].passed) automatedREarned += testMarks;







          }







          continue;







        }







        objectiveTotal++;







        var chosen = answer.chosen !== undefined ? answer.chosen : answer.chosenText;







        var ok = isObjectiveAnswerCorrect_(chosen, key);







        if (ok) objectiveCorrect++;







        if (!answers[i]) answers[i] = {};







        answers[i].q = answers[i].q || key.q;







        answers[i].question = answers[i].question || key.q;







        answers[i].type = key.type || "mcq";







        answers[i].blanks = key.blanks;







        answers[i].options = key.options;







        answers[i].correct = key.correct;







        answers[i].correctText = answerTextFromKey_(key);







        answers[i].isCorrect = ok;







      }







      var automaticEarned = objectiveCorrect + automatedREarned;







      var automaticMax = objectiveTotal + automatedRMax;







      var score = automaticMax ? Math.round(automaticEarned / automaticMax * 100) : 0;







      var grading = row[8] ? safeJsonParse_(row[8], {}) : {};







      var gradingJson = row[8] || "";







      var finalScore = row[9];







      if (grading && Object.keys(grading).length) {







        var gradingQuestions = grading.questions || {};







        var rEarned = 0;







        var rMax = 0;







        for (var ri = 0; ri < keyQuestions.length; ri++) {







          var rKey = keyQuestions[ri] || {};







          if (rKey.type !== "rcoding" && rKey.type !== "r coding") continue;







          var rAnswer = answers[ri] || {};







          var rDefinitions = Array.isArray(rKey.tests) ? rKey.tests : [];







          var rResults = rAnswer.automated && Array.isArray(rAnswer.automated.tests) ? rAnswer.automated.tests : [];







          var questionAutoEarned = 0;







          var questionMax = 0;







          for (var rti = 0; rti < rDefinitions.length; rti++) {







            var rMarks = Number(rDefinitions[rti].marks) || 0;







            questionMax += rMarks;







            if (rResults[rti] && rResults[rti].passed) questionAutoEarned += rMarks;







          }







          var override = gradingQuestions[ri];







          if (override) {







            var manualValue = override.manualScore;







            if (manualValue === undefined && Array.isArray(override.rubricScores)) {







              manualValue = override.rubricScores.reduce(function(sum, value) { return sum + (Number(value) || 0); }, 0);







            }







            rEarned += Math.max(0, Math.min(questionMax, Number(manualValue) || 0));







          } else {







            rEarned += questionAutoEarned;







          }







          rMax += questionMax;







        }







        var overallEarned = objectiveCorrect + rEarned;







        var overallMax = objectiveTotal + rMax;







        grading.objectiveEarned = objectiveCorrect;







        grading.objectiveMax = objectiveTotal;







        grading.overallEarned = overallEarned;







        grading.overallMax = overallMax;







        grading.earned = overallEarned;







        grading.max = overallMax;







        grading.finalScore = overallMax ? Math.round(overallEarned / overallMax * 100) : 0;







        grading.hasManualOverrides = Object.keys(gradingQuestions).length > 0;







        grading.recalculatedAt = new Date().toISOString();







        gradingJson = JSON.stringify(grading);







        finalScore = grading.finalScore;







      }







      responseSheet.getRange(r + 1, 4, 1, 7).setValues([[







        score + "%",







        objectiveCorrect,







        objectiveTotal,







        row[6],







        JSON.stringify(answers),







        gradingJson,







        finalScore === undefined || finalScore === null ? "" : finalScore







      ]]);







      updated++;







    }







    return { success: true, updated: updated, skipped: skipped, examTitle: activeTitle };







  } catch(e) {







    return { success: false, error: e.message };







  }







}







function saveRubricGrade(rowNumber, payload) {







  try {







    var ss = getOrCreateSheet();







    var sheet = ss.getSheetByName("Responses");







    rowNumber = parseInt(rowNumber, 10);







    if (!sheet || rowNumber < 2 || rowNumber > sheet.getLastRow()) {







      return { success: false, error: "Response row not found." };







    }







    ensureResponseHeaders_(sheet);







    payload = payload || {};







    if (payload.hasManualOverrides) {







      sheet.getRange(rowNumber, 9).setValue(JSON.stringify(payload));







      sheet.getRange(rowNumber, 10).setValue(payload.finalScore === undefined ? "" : payload.finalScore);







    } else {







      // No manual override: revert fully to the stored automated score.







      sheet.getRange(rowNumber, 9, 1, 2).clearContent();







    }







    return { success: true };







  } catch(e) {







    return { success: false, error: e.message };







  }







}







function clearResponses() {







  try {







    var ss = getOrCreateSheet();







    var sheet = ss.getSheetByName("Responses");







    if (!sheet) return { success: true };















    var lastRow = sheet.getLastRow();







    if (lastRow > 1) {







      sheet.deleteRows(2, lastRow - 1);







    }







    return { success: true };







  } catch(e) {







    return { success: false, error: e.message };







  }







}

function uploadImage(base64Data, filename, mimeType) {







  try {







    var folderName = "Exam Images";







    var folders = DriveApp.getFoldersByName(folderName);







    var folder;















    // Find or create the folder







    if (folders.hasNext()) {







      folder = folders.next();







    } else {







      folder = DriveApp.createFolder(folderName);







      // Set sharing so students can view the images







      folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);







    }















    var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), mimeType, filename);







    var file = folder.createFile(blob);















    // RETURN THE RAW THUMBNAIL URL INSTEAD OF THE PREVIEW PAGE







    return {







      success: true,







      url: "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1000"







    };







  } catch(e) {







    return { success: false, error: e.message };







  }







}
