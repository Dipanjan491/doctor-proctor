For questions with <strong>more than one blank</strong> (up to 4), fill in <strong>Correct Answer 2</strong>, <strong>Correct Answer 3</strong>, and/or <strong>Correct Answer 4</strong> with the accepted answer(s) for each additional blank, in order. function responseHasR(r){return Array.isArray(r&&r.answers)&&(r.answers.some(a=>a&&(a.type==='rcoding'||a.type==='r coding')));}
function responseHasManualOverride(r){return !!(r&&((r.finalScore!==null&&r.finalScore!==undefined&&Number.isFinite(Number(r.finalScore)))||(r.grading&&r.grading.hasManualOverrides)));}
function responseNeedsAttention(r){return responseHasR(r)&&(r.answers||[]).filter(a=>a&&(a.type==='rcoding'||a.type==='r coding')).some(a=>getAutomatedQuestionScore(a).maximum<=0);}
function responseGradingState(r){if(responseHasManualOverride(r))return {label:'Score adjusted',kind:'info'};if(responseNeedsAttention(r))return {label:'Needs attention',kind:'warning'};if(responseHasR(r))return {label:'Review available',kind:'neutral'};return {label:'Automated',kind:'success'};}
function scoreBand(score){return score>=70?{label:'Pass',kind:'success'}:(score>=50?{label:'Borderline',kind:'warning'}:{label:'Below 50%',kind:'danger'});}
function pill(label,kind){return '<span class="status-pill '+kind+'"><span class="status-dot-small"></span>'+escapeHtml(label)+'</span>';}

async function loadDashboard(){

const active=results[0].status==='fulfilled'?results[0].value:null;
const responses=results[1].status==='fulfilled'?(results[1].value||[]):[];

const sets=results[3].status==='fulfilled'?(results[3].value||[]):[];
if(active)applyActiveExamUI(active);else renderDashboardExam(null);
if(results[1].status==='fulfilled')responsesData=responses;
if(results[2].status==='fulfilled')saData=security;

}catch(err){showToast('Dashboard could not be loaded. Please refresh and try again.','error',0);}finally{setAppLoading(false);}
}

function renderDashboardData(responses,security,sets){
const total=responses.length;const average=total?Math.round(responses.reduce((sum,r)=>sum+responseScore(r),0)/total):0;
const review=responses.filter(r=>responseHasR(r)&&!responseHasManualOverride(r)).length;
const high=security.filter(r=>r.riskLevel==='high').length;
['dash-submissions','dash-average','dash-review','dash-high-risk','dash-saved-sets'].forEach(id=>document.getElementById(id).classList.remove('skeleton'));document.getElementById('dash-submissions').textContent=total;document.getElementById('dash-average').textContent=average+'%';document.getElementById('dash-review').textContent=review;document.getElementById('dash-high-risk').textContent=high;document.getElementById('dash-saved-sets').textContent=sets.length;
document.getElementById('dash-submissions-note').textContent=total===1?'1 recorded response':total+' recorded responses';document.getElementById('dash-risk-note').textContent=high?high+' require attention':'No high-risk records';
setNavCount('nav-responses-count',total);setNavCount('nav-security-count',high);setNavCount('nav-sets-count',sets.length);
renderDashboardRecent(responses.slice(0,5));renderDashboardSecurity(security.slice().sort((a,b)=>(b.riskScore||0)-(a.riskScore||0)).slice(0,5));
}

function renderDashboardRecent(rows){
const body=document.getElementById('dash-recent-body');if(!body)return;
if(!rows.length){body.innerHTML='<tr><td colspan="5" class="dashboard-empty">No submissions have been received yet.</td></tr>';return;}
body.innerHTML=rows.map(r=>{const score=responseScore(r),grading=responseGradingState(r);return '<tr><td><strong>'+escapeHtml(r.name||'Anonymous')+'</strong></td><td>'+escapeHtml(r.examTitle||'')+'</td><td>'+score+'% '+pill(scoreBand(score).label,scoreBand(score).kind)+'</td><td>'+pill(grading.label,grading.kind)+'</td><td><button class="btn btn-ghost btn-sm" onclick="openDashboardResponse('+Number(r.rowNumber)+')">Open</button></td></tr>';}).join('');
}

function renderDashboardSecurity(rows){

function parseRTests(raw){
return raw.split(/\r?\n/).map(s=>s.trim()).filter(Boolean).map((line,i)=>{
const parts=line.split('|');if(parts.length<3)throw new Error('Automated test line '+(i+1)+' must use: Label | marks | R expression');
return {label:parts.shift().trim(),marks:Number(parts.shift().trim())||0,expression:parts.join('|').trim()};
});
}

function parseRubric(raw){
return raw.split(/\r?\n/).map(s=>s.trim()).filter(Boolean).map(line=>{
const parts=line.split('|');
if(parts.length<2)return {criterion:line,marks:0};
return {criterion:parts.shift().trim(),marks:Number(parts.shift().trim())||0};
});
}

function getBlankGroups(q){

const groups=(q.blanks>=2 && Array.isArray(q.correct) && Array.isArray(q.correct[0]))
? q.correct
: [Array.isArray(q.correct)?q.correct:[q.correct]];

function bracketBalance(text){

let depth=0,quote='',escaped=false;

for(const ch of String(text||'')){

if(escaped){escaped=false;continue;}

if(ch==='\\'){escaped=true;continue;}

if(quote){if(ch===quote)quote='';continue;}

if(ch==='"'||ch==="'"){quote=ch;continue;}

if(ch==='('||ch==='['||ch==='{')depth++;

else if(ch===')'||ch===']'||ch==='}')depth--;

}

return depth;

}

function repairLegacyCommaSplitAnswers(answers){

const source=Array.isArray(answers)?answers:[answers];

const repaired=[];let pending='',depth=0;

source.forEach(value=>{

const part=String(value===null||value===undefined?'':value).trim();

if(!part)return;

pending=pending?pending+', '+part:part;

depth+=bracketBalance(part);

if(depth<=0){repaired.push(pending);pending='';depth=0;}

});

if(pending)repaired.push(pending);

return repaired;

}

function splitTopLevelCommaAlternatives(text){

const parts=[];let current='',depth=0,quote='',escaped=false;

for(const ch of String(text||'')){

if(escaped){current+=ch;escaped=false;continue;}

if(ch==='\\'){current+=ch;escaped=true;continue;}

if(quote){current+=ch;if(ch===quote)quote='';continue;}

if(ch==='"'||ch==="'"){quote=ch;current+=ch;continue;}

if(ch==='('||ch==='['||ch==='{'){depth++;current+=ch;continue;}

if(ch===')'||ch===']'||ch==='}'){depth=Math.max(0,depth-1);current+=ch;continue;}

if(ch===','&&depth===0){if(current.trim())parts.push(current.trim());current='';continue;}

current+=ch;

}

if(current.trim())parts.push(current.trim());

return parts;

}

function parseAcceptedAnswers(raw){

const text=String(raw||'').trim();

if(!text)return [];

if(text.includes('||'))return text.split(/\s*\|\|\s*/).map(s=>s.trim()).filter(Boolean);

return splitTopLevelCommaAlternatives(text);

}

function parseMcqCorrectAnswers(raw){

const letterMap={A:0,B:1,C:2,D:3};

const indexes=String(raw||'').toUpperCase().split(/[\s,;|/]+/).map(token=>letterMap[token]).filter(index=>index!==undefined);

return [...new Set(indexes)].sort((a,b)=>a-b);

}

function exportSheet(wb,name,rows,widths){
const ws=XLSX.utils.aoa_to_sheet(rows);
const range=XLSX.utils.decode_range(ws['!ref']||'A1:A1');
for(let c=range.s.c;c<=range.e.c;c++){
const ref=XLSX.utils.encode_cell({r:0,c});
if(ws[ref])ws[ref].s={font:{bold:true,color:{rgb:'FFFFFF'}},fill:{fgColor:{rgb:'1E3A5F'}},alignment:{vertical:'center',wrapText:true}};
}
ws['!cols']=widths||autoFitColumns(rows,0,12,48);
ws['!rows']=[{hpt:26}];
if(rows.length>1)ws['!autofilter']={ref:XLSX.utils.encode_range({s:{r:0,c:0},e:{r:Math.max(0,rows.length-1),c:range.e.c}})};
XLSX.utils.book_append_sheet(wb,ws,name);
return ws;
}

function uniqueSheetName(wb,raw){
const base=(raw||'Dataset').replace(/[\[\]:*?\/\\]/g,' ').trim().slice(0,31)||'Dataset';
let name=base,n=2;
while(wb.SheetNames.includes(name)){const suffix=' '+n++;name=base.slice(0,31-suffix.length)+suffix;}
return name;
}

function processFile(file){

showStatus('upload-status','Reading file...','success');

const reader=new FileReader();

reader.onload=function(e){

try{

let parsed=[];

const isCsv=file.name.toLowerCase().endsWith('.csv');

const wb=XLSX.read(e.target.result,{type:isCsv?'string':'binary'});

const questionSheetName=wb.SheetNames.find(name=>normaliseExcelHeader(name)==='questions')||wb.SheetNames[0];

function normaliseExcelHeader(value){
return String(value||'').trim().toLowerCase().replace(/[^a-z0-9]+/g,'');
}

function findExcelColumn(headers,names,fallback){
const wanted=names.map(normaliseExcelHeader);
for(let i=0;i<headers.length;i++)if(wanted.includes(normaliseExcelHeader(headers[i])))return i;
return fallback;
}

function parseRows(rows,wb){

const res=[];

if(!Array.isArray(rows)||!rows.length)return res;

const headers=rows[0]||[];

const columns={
question:findExcelColumn(headers,['Question'],0),
optionA:findExcelColumn(headers,['Option A'],1),
optionB:findExcelColumn(headers,['Option B'],2),
optionC:findExcelColumn(headers,['Option C'],3),
optionD:findExcelColumn(headers,['Option D'],4),
correct:findExcelColumn(headers,['Correct Answer'],5),
part:findExcelColumn(headers,['Part','Section'],6),
type:findExcelColumn(headers,['Question Type','Type'],7),
correct2:findExcelColumn(headers,['Correct Answer 2'],-1),
correct3:findExcelColumn(headers,['Correct Answer 3'],-1),
correct4:findExcelColumn(headers,['Correct Answer 4'],-1),
starter:findExcelColumn(headers,['Starter Code'],-1),
limit:findExcelColumn(headers,['Execution Limit (s)','Execution Limit','Execution Limit Per Run (seconds)'],-1),
packages:findExcelColumn(headers,['R Packages','Packages'],-1),
tests:findExcelColumn(headers,['Automated Tests','Automatic Tests','Test Definitions'],-1),
rubric:findExcelColumn(headers,['Manual Review Guidance','Review Guidance','Manual Rubric'],-1)
};

const supplementalTests=readSupplementalAutomatedTests(wb);

const supplementalRubric=readSupplementalRubric(wb);

const cell=(row,index)=>index>=0?row[index]:'';

for(let i=1;i<rows.length;i++){

const r=rows[i];if(!r||!String(cell(r,columns.question)||'').trim())continue;

const q=String(cell(r,columns.question)||'').trim(),a=String(cell(r,columns.optionA)||'').trim(),b=String(cell(r,columns.optionB)||'').trim(),c=String(cell(r,columns.optionC)||'').trim(),d=String(cell(r,columns.optionD)||'').trim();

const correctRaw=String(cell(r,columns.correct)||'').trim(),part=String(cell(r,columns.part)||'').trim()||'General';

const typeRaw=String(cell(r,columns.type)||'').trim().toLowerCase();

const isFillBlank=typeRaw==='fillblank'||typeRaw==='fill in the blank'||typeRaw==='fill-in-the-blank';

const isRcoding=typeRaw==='rcoding'||typeRaw==='r coding';

if(!q)continue;

if (isRcoding) {

try {

const inlineTests=String(cell(r,columns.tests)||'').trim();

const inlineRubric=String(cell(r,columns.rubric)||'').trim();

const questionNumber=String(i);

const tests=inlineTests?parseRTests(inlineTests):(supplementalTests.byNumber[questionNumber]||supplementalTests.byQuestion[q]||[]);

const rubric=inlineRubric?parseRubric(inlineRubric):(supplementalRubric.byNumber[questionNumber]||supplementalRubric.byQuestion[q]||[]);

res.push({

part,

q,

type:'rcoding',

correct:'Automated and rubric grading',

starterCode:String(cell(r,columns.starter)||''),

executionLimit:Math.max(1,Math.min(60,Number(cell(r,columns.limit))||10)),

packages:String(cell(r,columns.packages)||'').trim(),

tests,

rubric

});

} catch(err) {

throw new Error('Excel row '+(i+1)+': '+err.message);

}

} else if(isFillBlank){

const accepted=parseAcceptedAnswers(correctRaw);

if(!accepted.length)continue;

const groups=[accepted];

for(const ci2 of [columns.correct2,columns.correct3,columns.correct4]){

if(ci2<0)break;

const raw=String(cell(r,ci2)||'').trim();

if(!raw)break;

const acc=parseAcceptedAnswers(raw);

if(!acc.length)break;

groups.push(acc);

}

if(groups.length===1) res.push({part,q,type:'fillblank',blanks:1,correct:groups[0]});

else res.push({part,q,type:'fillblank',blanks:groups.length,correct:groups});

}else{

if(!a||!b)continue;

const correctIndexes=parseMcqCorrectAnswers(correctRaw);if(!correctIndexes.length)continue;

res.push({part,q,type:'mcq',options:[a,b,c,d],correct:correctIndexes.length===1?correctIndexes[0]:correctIndexes});

}

}

return res;

}

function getQuestionTestDefinitions(answer){
const stored=Array.isArray(answer&&answer.testDefinitions)?answer.testDefinitions:[];
if(stored.length)return stored;
const allKnownQuestions=[...(questions||[]),...(activeQuestionsCache||[]),...savedSetsCache.flatMap(set=>(set&&set.data&&Array.isArray(set.data.questions))?set.data.questions:[])];
const current=allKnownQuestions.find(q=>q&&(q.type==='rcoding'||q.type==='r coding')&&String(q.q||'').trim()===String(answer&&answer.question||'').trim());
if(current&&Array.isArray(current.tests)&&current.tests.length)return current.tests;
return (answer&&answer.automated&&Array.isArray(answer.automated.tests))?answer.automated.tests:[];
}

function getAutomatedQuestionScore(answer){
const definitions=getQuestionTestDefinitions(answer);
const results=(answer&&answer.automated&&Array.isArray(answer.automated.tests))?answer.automated.tests:[];
const maximum=definitions.reduce((sum,test)=>sum+(Number(test.marks)||0),0);
const earned=definitions.reduce((sum,test,index)=>sum+(results[index]&&results[index].passed?(Number(test.marks)||0):0),0);
return {definitions,results,earned,maximum};
}

function savedManualOverrideValue(saved){
if(!saved)return '';
if(saved.manualOverride===true&&saved.manualScore!==undefined)return saved.manualScore;
if(Array.isArray(saved.rubricScores))return saved.rubricScores.reduce((sum,value)=>sum+(Number(value)||0),0);
if(saved.manualScore!==undefined)return saved.manualScore;
return '';
}

// Helper: Dynamically calculates column widths based on maximum text length

function autoFitColumns(rows, startRowIdx = 0, minW = 14, maxW = 50) {

if (!rows || !rows.length) return [];

let colCount = 0;

rows.forEach(r => { if (r && r.length > colCount) colCount = r.length; });

const cols = [];

for (let c = 0; c < colCount; c++) {

let maxLen = 0;

for (let r = startRowIdx; r < rows.length; r++) {

const val = rows[r] ? rows[r][c] : "";

if (val !== null && val !== undefined && String(val).trim() !== "") {

const lines = String(val).split('\n');

lines.forEach(line => {

if (line.length > maxLen) maxLen = line.length;

});

}

}

cols.push({ wch: Math.min(Math.max(maxLen + 4, minW), maxW) });

}

return cols;

}
